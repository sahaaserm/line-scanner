function submitData() { let code = document.getElementById("barcodeResult").value; let qty = document.getElementById("qtyInput").value; let price = document.getElementById("priceInput").value; if (!code) { alert("⚠️ กรุณาสแกนบาร์โค้ดให้สำเร็จก่อนครับ"); return; } if (!qty || qty <= 0) { alert("⚠️ กรุณาระบุจำนวนสินค้าให้ถูกต้อง"); return; } if (!price || price < 0) { alert("⚠️ กรุณาระบุราคารวมจ่ายให้ถูกต้อง"); return; } // รวมข้อความให้อยู่ในรูปแบบ: รหัสบาร์โค้ด*จำนวน*ยอดจ่าย let textMessage = `${code}*${qty}*${price}`; // เช็กว่าเปิดผ่านแอป LINE หรือไม่ if (liff.isInClient()) { liff.sendMessages([ { type: 'text', text: textMessage } ]).then(() => { liff.closeWindow(); // ส่งเสร็จแล้วปิด LIFF ทันที }).catch((err) => { // แสดงกล่องแจ้งเตือน Error ที่เกิดขึ้น alert("❌ ส่งเข้าแชตไม่สำเร็จ เกิดจาก: " + err.code + " - " + err.message); }); } else { // หากทดสอบเปิดผ่าน Browser คอมพิวเตอร์/ปกติ alert("📌 ข้อความที่จะส่งเข้าแชตคือ: " + textMessage + "\n(หมายเหตุ: การส่งเข้าแชตอัตโนมัติกดได้เฉพาะเมื่อเปิดผ่านแอป LINE เท่านั้น)"); } }
